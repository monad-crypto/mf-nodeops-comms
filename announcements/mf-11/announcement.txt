Example of Kernel upgrade:

# Set Kernel version
V=6.8.0-149-generic

# Install the target kernel packages (the apt task, including update_cache)
apt-get update
apt-get install -y linux-image-$V linux-headers-$V linux-modules-$V linux-modules-extra-$V linux-tools-$V

# Remove the generic metapackages (state: absent = remove, not purge)
apt-get remove -y --allow-change-held-packages linux-image-generic linux-headers-generic

# Set the GRUB_DEFAULT to the new kernel and reboot
sed -i "s|^GRUB_DEFAULT=.*|GRUB_DEFAULT=\"Advanced options for Ubuntu>Ubuntu, with Linux $V\"|" /etc/default/grub
update-grub
reboot
